import express from 'express'; import path from 'node:path'; import { readFileSync, existsSync } from 'node:fs'; import { fileURLToPath } from 'node:url'; import multer from 'multer'; import { config } from './config.js'; import { applySecurityMiddleware, apiLimiter } from './security.js'; import { verifyAuth, login } from './auth.js'; import { createSession, destroySession, getSessionFromToken } from './storage.js'; import { listConversations, createConversation, getConversation, addMessageToConversation, updateConversationTitle, deleteConversation, searchConversations, exportConversation, } from './services/conversationService.js'; import { addMemory, clearMemories, deleteMemory, listMemories } from './services/memoryService.js'; import { callLlm } from './services/llmProvider.js'; import { runResearch } from './services/researchService.js'; import { extractTextFromFile, safeErrorMessage, sanitizeUserInput, validateTextFileType } from './services/fileProcessor.js'; const app = express(); const upload = multer({ storage: multer.memoryStorage(), limits: { fileSize: (config.maxUploadSizeMb || 10) * 1024 * 1024 }, }); const __filename = fileURLToPath(import.meta.url); const __dirname = path.dirname(__filename); const publicDir = path.join(__dirname, '../../public'); app.use(express.json({ limit: '2mb' })); applySecurityMiddleware(app); app.use('/api', apiLimiter); app.get('/api/health', (req, res) => { res.json({ status: 'ok', app: config.appName, mode: config.nodeEnv }); }); app.post('/api/auth/login', (req, res) => { const username = String(req.body.username ?? ''); const password = String(req.body.password ?? ''); if (username === config.defaultUsername && password === config.defaultPassword) { const token = createSession(username); res.json({ token, user: { username } }); return; } res.status(401).json({ error: 'Invalid credentials' }); }); app.post('/api/auth/logout', (req, res) => { const authHeader = req.headers.authorization ?? ''; const token = authHeader.replace('Bearer ', '').trim(); destroySession(token); res.json({ success: true }); }); app.get('/api/conversations', verifyAuth, (req, res) => { res.json(listConversations()); }); app.post('/api/conversations', verifyAuth, (req, res) => { const title = String(req.body.title ?? 'New conversation'); const conversation = createConversation(title); res.status(201).json(conversation); }); app.get('/api/conversations/:id', verifyAuth, (req, res) => { const conversation = getConversation(req.params.id); if (!conversation) { res.status(404).json({ error: 'Conversation not found' }); return; } res.json(conversation); }); app.patch('/api/conversations/:id', verifyAuth, (req, res) => { const title = String(req.body.title ?? '').trim(); if (!title) { res.status(400).json({ error: 'Title is required.' }); return; } const conversation = updateConversationTitle(req.params.id, title); if (!conversation) { res.status(404).json({ error: 'Conversation not found' }); return; } res.json(conversation); }); app.delete('/api/conversations/:id', verifyAuth, (req, res) => { const deleted = deleteConversation(req.params.id); res.json({ success: deleted }); }); app.get('/api/conversations/search', verifyAuth, (req, res) => { const query = String(req.query.q ?? ''); res.json(searchConversations(query)); }); app.get('/api/conversations/:id/export', verifyAuth, (req, res) => { try { const content = exportConversation(req.params.id); res.setHeader('Content-Type', 'application/json'); res.setHeader('Content-Disposition', `attachment; filename="${req.params.id}.json"`); res.send(content); } catch (error) { res.status(404).json({ error: safeErrorMessage(error) }); } }); app.post('/api/chat', verifyAuth, async (req, res) => { const userPrompt = sanitizeUserInput(String(req.body.prompt ?? '')); const conversationId = String(req.body.conversationId ?? ''); if (!userPrompt) { res.status(400).json({ error: 'Prompt is required.' }); return; } let conversation = conversationId ? getConversation(conversationId) : undefined; if (!conversation) { conversation = createConversation('New conversation'); } addMessageToConversation(conversation.id, 'user', userPrompt); const history = conversation.messages .slice(-8) .map((message) => ({ role: message.role === 'assistant' ? 'assistant' : 'user', content: message.content })); try { const llmResponse = await callLlm(history); const assistantMessage = addMessageToConversation(conversation.id, 'assistant', llmResponse.content); res.json({ conversationId: conversation.id, message: assistantMessage, usage: llmResponse.usage }); } catch (error) { const fallback = safeErrorMessage(error); const assistantMessage = addMessageToConversation(conversation.id, 'assistant', `Unable to generate a response. ${fallback}`); res.status(500).json({ error: fallback, conversationId: conversation.id, message: assistantMessage }); } }); app.post('/api/chat/stream', verifyAuth, async (req, res) => { res.setHeader('Content-Type', 'text/event-stream'); res.setHeader('Cache-Control', 'no-cache, no-transform'); res.setHeader('Connection', 'keep-alive'); const prompt = sanitizeUserInput(String(req.body.prompt ?? '')); if (!prompt) { res.write(`data: ${JSON.stringify({ error: 'Prompt is required.' })}\n\n`); res.end(); return; } try { const llmResponse = await callLlm([{ role: 'user', content: prompt }], { stream: true }); const chunks = llmResponse.content.split(/\s+/); for (const chunk of chunks) { res.write(`data: ${JSON.stringify({ type: 'chunk', text: `${chunk} ` })}\n\n`); } res.write(`data: ${JSON.stringify({ type: 'done' })}\n\n`); } catch (error) { res.write(`data: ${JSON.stringify({ error: safeErrorMessage(error) })}\n\n`); } res.end(); }); app.post('/api/research', verifyAuth, async (req, res) => { const question = sanitizeUserInput(String(req.body.question ?? '')); if (!question) { res.status(400).json({ error: 'Research question is required.' }); return; } const result = await runResearch(question); res.json(result); }); app.get('/api/memory', verifyAuth, (req, res) => { res.json({ memories: listMemories() }); }); app.post('/api/memory', verifyAuth, (req, res) => { const content = sanitizeUserInput(String(req.body.content ?? '')); const source = String(req.body.source ?? 'user'); const consent = Boolean(req.body.consent ?? true); if (!content) { res.status(400).json({ error: 'Memory content is required.' }); return; } const memory = addMemory(content, source, consent); res.status(201).json(memory); }); app.delete('/api/memory/:id', verifyAuth, (req, res) => { deleteMemory(req.params.id); res.json({ success: true }); }); app.post('/api/memory/clear', verifyAuth, (req, res) => { clearMemories(); res.json({ success: true }); }); app.post('/api/files/analyze', verifyAuth, upload.single('file'), async (req, res) => { const file = req.file; if (!file) { res.status(400).json({ error: 'Please upload a supported file.' }); return; } if (!validateTextFileType(file)) { res.status(400).json({ error: 'Unsupported file type. Supported: PDF, TXT, CSV, DOCX, PNG, JPG, WEBP.' }); return; } try { const result = await extractTextFromFile(file); res.json({ success: true, result }); } catch (error) { res.status(500).json({ error: safeErrorMessage(error) }); } }); app.get('/api/settings', verifyAuth, (req, res) => { res.json({ provider: 'openai-compatible', model: config.openAiModel, baseUrl: config.openAiBaseUrl || 'https://api.openai.com/v1', searchEngine: config.searchEngine, language: 'en', }); }); app.post('/api/settings', verifyAuth, (req, res) => { res.json({ success: true, message: 'Settings saved to server-side environment variables.' }); }); app.get('/api/session', verifyAuth, (req, res) => { const token = req.headers.authorization?.replace('Bearer ', '').trim(); const session = getSessionFromToken(token); res.json({ user: session ? { username: session.username } : null }); }); const rootFile = path.join(publicDir, 'index.html'); if (existsSync(rootFile)) { app.use(express.static(publicDir)); app.get('*', (req, res) => { if (!req.path.startsWith('/api')) { res.sendFile(rootFile); } }); } export { app }; if (config.nodeEnv !== 'test') { app.listen(config.port, () => { console.log(`SHADOW AI running on http://localhost:${config.port}`); }); }

import express from 'express'; import path from 'node:path'; import { existsSync } from 'node:fs'; import multer from 'multer'; import pdfParse from 'pdf-parse'; import mammoth from 'mammoth'; import { config } from './config.js'; import { login, requireAuth } from './auth.js'; import { applySecurity, apiLimiter } from './security.js'; import { loadStore, saveStore } from './dataStore.js'; import { callLLM, searchWeb } from './llm.js'; import { makeId, sanitizeText, safeErrorMessage, buildResearchSummary } from './utils.js'; const app = express(); const upload = multer({ storage: multer.memoryStorage(), limits: { fileSize: (config.MAX_UPLOAD_SIZE_MB ?? 10) * 1024 * 1024 }, }); app.use(express.json({ limit: '2mb' })); applySecurity(app); app.use('/api', apiLimiter); app.get('/api/health', (_req, res) => { res.json({ status: 'ok', service: config.APP_NAME, environment: config.NODE_ENV }); }); app.post('/api/auth/login', (req, res) => { const username = String(req.body.username ?? ''); const password = String(req.body.password ?? ''); const result = login(username, password); if (!result) { res.status(401).json({ error: 'Invalid credentials' }); return; } res.json(result); }); app.post('/api/auth/logout', requireAuth, (req, res) => { res.json({ success: true, user: req.user }); }); app.get('/api/conversations', requireAuth, (_req, res) => { const store = loadStore(); const sorted = [...store.conversations].sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime()); res.json(sorted); }); app.post('/api/conversations', requireAuth, (req, res) => { const store = loadStore(); const conversation = { id: makeId('conversation'), title: sanitizeText(String(req.body.title ?? 'New conversation')) || 'New conversation', createdAt: new Date().toISOString(), updatedAt: new Date().toISOString(), messages: [], }; store.conversations.unshift(conversation); saveStore(store); res.status(201).json(conversation); }); app.get('/api/conversations/:id', requireAuth, (req, res) => { const store = loadStore(); const conversation = store.conversations.find((item) => item.id === req.params.id); if (!conversation) { res.status(404).json({ error: 'Conversation not found' }); return; } res.json(conversation); }); app.patch('/api/conversations/:id', requireAuth, (req, res) => { const store = loadStore(); const conversation = store.conversations.find((item) => item.id === req.params.id); if (!conversation) { res.status(404).json({ error: 'Conversation not found' }); return; } const title = sanitizeText(String(req.body.title ?? '')); if (!title) { res.status(400).json({ error: 'Title is required.' }); return; } conversation.title = title; conversation.updatedAt = new Date().toISOString(); saveStore(store); res.json(conversation); }); app.delete('/api/conversations/:id', requireAuth, (req, res) => { const store = loadStore(); const before = store.conversations.length; store.conversations = store.conversations.filter((item) => item.id !== req.params.id); saveStore(store); res.json({ success: store.conversations.length !== before }); }); app.get('/api/conversations/search', requireAuth, (req, res) => { const query = sanitizeText(String(req.query.q ?? '')).toLowerCase(); const store = loadStore(); const results = store.conversations.filter((conversation) => { const haystack = `${conversation.title} ${conversation.messages.map((message) => message.content).join(' ')}`.toLowerCase(); return haystack.includes(query); }); res.json(results); }); app.get('/api/conversations/:id/export', requireAuth, (req, res) => { const store = loadStore(); const conversation = store.conversations.find((item) => item.id === req.params.id); if (!conversation) { res.status(404).json({ error: 'Conversation not found' }); return; } res.setHeader('Content-Type', 'application/json'); res.setHeader('Content-Disposition', `attachment; filename="${conversation.id}.json"`); res.send(JSON.stringify(conversation, null, 2)); }); app.post('/api/chat', requireAuth, async (req, res) => { const prompt = sanitizeText(String(req.body.prompt ?? '')); if (!prompt) { res.status(400).json({ error: 'Prompt is required.' }); return; } const store = loadStore(); const conversationId = String(req.body.conversationId ?? ''); let conversation = store.conversations.find((item) => item.id === conversationId); if (!conversation) { conversation = { id: makeId('conversation'), title: 'New conversation', createdAt: new Date().toISOString(), updatedAt: new Date().toISOString(), messages: [], }; store.conversations.unshift(conversation); } const userMessage = { id: makeId('msg'), role: 'user' as const, content: prompt, createdAt: new Date().toISOString(), }; conversation.messages.push(userMessage); conversation.updatedAt = new Date().toISOString(); try { const llmPayload = await callLLM( conversation.messages.slice(-8).map((message) => ({ role: message.role === 'assistant' ? 'assistant' : 'user', content: message.content, })) ); const assistantMessage = { id: makeId('msg'), role: 'assistant' as const, content: llmPayload.content, createdAt: new Date().toISOString(), }; conversation.messages.push(assistantMessage); conversation.updatedAt = new Date().toISOString(); saveStore(store); res.json({ conversationId: conversation.id, message: assistantMessage, usage: llmPayload.usage }); } catch (error) { const fallback = `The assistant could not generate a response. ${safeErrorMessage(error)}`; const assistantMessage = { id: makeId('msg'), role: 'assistant' as const, content: fallback, createdAt: new Date().toISOString(), }; conversation.messages.push(assistantMessage); saveStore(store); res.status(500).json({ error: fallback, message: assistantMessage }); } }); app.post('/api/research', requireAuth, async (req, res) => { const question = sanitizeText(String(req.body.question ?? '')); if (!question) { res.status(400).json({ error: 'Research question is required.' }); return; } try { const sources = await searchWeb(question); const summary = buildResearchSummary(question, sources); res.json({ question, summary, sources }); } catch (error) { res.status(500).json({ error: safeErrorMessage(error) }); } }); app.get('/api/memory', requireAuth, (_req, res) => { const store = loadStore(); res.json({ memories: store.memories }); }); app.post('/api/memory', requireAuth, (req, res) => { const content = sanitizeText(String(req.body.content ?? '')); if (!content) { res.status(400).json({ error: 'Memory content is required.' }); return; } const store = loadStore(); const memory = { id: makeId('memory'), content, source: String(req.body.source ?? 'user'), consent: Boolean(req.body.consent ?? true), createdAt: new Date().toISOString(), }; store.memories.unshift(memory); saveStore(store); res.status(201).json(memory); }); app.delete('/api/memory/:id', requireAuth, (req, res) => { const store = loadStore(); const before = store.memories.length; store.memories = store.memories.filter((entry) => entry.id !== req.params.id); saveStore(store); res.json({ success: store.memories.length !== before }); }); app.post('/api/memory/clear', requireAuth, (_req, res) => { const store = loadStore(); store.memories = []; saveStore(store); res.json({ success: true }); }); app.post('/api/files/analyze', requireAuth, upload.single('file'), async (req, res) => { const file = req.file; if (!file) { res.status(400).json({ error: 'Please upload a file.' }); return; } const name = file.originalname.toLowerCase(); const allowed = ['.txt', '.csv', '.pdf', '.docx', '.png', '.jpg', '.jpeg', '.webp', '.gif']; const supported = allowed.some((ext) => name.endsWith(ext)) || file.mimetype.startsWith('image/'); if (!supported) { res.status(400).json({ error: 'Unsupported file type. Supported: TXT, CSV, PDF, DOCX, PNG, JPG, WEBP, GIF.' }); return; } try { let text = ''; if (file.mimetype === 'application/pdf' || name.endsWith('.pdf')) { const parsed = await pdfParse(file.buffer); text = parsed.text ?? ''; } else if (file.mimetype.includes('word') || name.endsWith('.docx')) { const result = await mammoth.extractRawText({ buffer: file.buffer }); text = result.value ?? ''; } else { text = file.buffer.toString('utf8'); } const normalized = sanitizeText(text, 12000); res.json({ success: true, file: { name: file.originalname, mimeType: file.mimetype, size: file.size, summary: normalized.slice(0, 500) || 'No readable content found.', text: normalized, }, }); } catch (error) { res.status(500).json({ error: safeErrorMessage(error) }); } }); app.get('/api/settings', requireAuth, (_req, res) => { res.json({ provider: 'openai-compatible', model: config.OPENAI_MODEL, baseUrl: config.OPENAI_BASE_URL, searchEngine: config.SEARCH_ENGINE, }); }); app.post('/api/settings', requireAuth, (_req, res) => { res.json({ success: true, message: 'Settings are configured on the server via environment variables.' }); }); const publicDir = path.join(process.cwd(), 'public'); const indexFile = path.join(publicDir, 'index.html'); if (existsSync(indexFile)) { app.use(express.static(publicDir)); app.get('*', (req, res) => { if (!req.path.startsWith('/api')) { res.sendFile(indexFile); } }); } export { app }; if (config.NODE_ENV !== 'test') { app.listen(config.PORT, () => { console.log(`SHADOW AI is running on http://localhost:${config.PORT}`); }); }
